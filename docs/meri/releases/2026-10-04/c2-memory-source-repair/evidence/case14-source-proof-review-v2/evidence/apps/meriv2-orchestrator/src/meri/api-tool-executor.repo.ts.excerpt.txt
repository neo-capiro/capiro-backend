42: function mailCapacityRetryDelay(toolName: string, response: Response, body: string): number | null {
43:   if (!RETRYABLE_MAIL_READS.has(toolName) || response.status !== 503) return null;
44:   let payload: unknown;
45:   try {
46:     payload = JSON.parse(body);
47:   } catch {
48:     return null;
49:   }
50:   if (
51:     !payload ||
52:     typeof payload !== 'object' ||
53:     (payload as Record<string, unknown>).statusCode !== 503 ||
54:     (payload as Record<string, unknown>).message !==
55:       'The database is temporarily busy. Please retry.'
56:   ) {
57:     return null;
58:   }
59:   const retryAfter = response.headers?.get('retry-after')?.trim();
60:   if (!retryAfter) return null;
61:   const delay = /^\d+$/.test(retryAfter)
62:     ? Number(retryAfter) * 1_000
63:     : Date.parse(retryAfter) - Date.now();
64:   // Honor the server's delay; do not shorten a long Retry-After and hammer it.
65:   return Number.isFinite(delay) && delay >= 0 && delay <= MAX_MAIL_RETRY_DELAY_MS
66:     ? Math.max(100, delay)
67:     : null;

70: function waitForMailRetry(delayMs: number, signal: AbortSignal): Promise<void> {
71:   return new Promise((resolve, reject) => {
72:     if (signal.aborted) {
73:       reject(signal.reason);

81:     const timer = setTimeout(() => {
82:       signal.removeEventListener('abort', onAbort);
83:       resolve();
84:     }, delayMs);

86:   });
