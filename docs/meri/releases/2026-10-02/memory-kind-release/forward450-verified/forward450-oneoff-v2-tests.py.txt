   if row['name'] in ('DB_USER','DB_PASSWORD'):row['valueFrom']+='-migration'

    if operation=='register_task_definition':result['taskDefinition']['taskDefinitionArn']=f'arn:aws:ecs:{m.REGION}:{m.ACCOUNT}:task-definition/{w.FAMILY}:1'
    return result
  aws=Aws();aws.logs_override=['unprefixed private log ignored']+[w.PREFIX+json.dumps(r) for r in self.receipts()];return aws

  return args,aws

 def test_database_role_endpoint_or_plaintext_shadow_refused(self):
  for key in ['DB_USER','DB_PASSWORD','DB_HOST']:
   original=copy.deepcopy(self.snapshot)
   for row in self.snapshot['migrationContainer']['secrets']:
    if row['name']==key:row['valueFrom']=next(x['valueFrom'] for x in self.snapshot['container']['secrets'] if x['name']==key) if key!='DB_HOST' else 'changed'
   with self.assertRaises(self.m.Refusal):self.request()
   self.snapshot=original
  self.snapshot['migrationContainer']['environment'].append({'name':'DB_USER','value':'not-a-secret'})
  with self.assertRaises(self.m.Refusal):self.request()
