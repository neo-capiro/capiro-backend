async function history(client,items){
 const rows=(await client.query('SELECT migration_name,checksum,finished_at IS NOT NULL AS finished,rolled_back_at IS NOT NULL AS rolled_back FROM _prisma_migrations ORDER BY migration_name')).rows;
 assert.equal(rows.length,items.length);assert.deepEqual(rows.map(x=>x.migration_name),items.map(x=>x.name));
 for(const row of rows){assert.equal(row.finished,true);assert.equal(row.rolled_back,false);assert.equal(items.find(x=>x.name===row.migration_name)?.sha256,row.checksum);}
 return {count:rows.length,exactSourceChecksums:true,unfinished:0,unexpected:0};
}
