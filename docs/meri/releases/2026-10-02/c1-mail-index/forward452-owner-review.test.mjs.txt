  ['legacy timestamp microsecond change', rows => { rows[0].started_at = '2026-06-27 12:36:39.855836+00'; }],
  ['legacy rolled-back row mutation', rows => { rows.find(row => row.rolled_back_at !== null).rolled_back_at = '2026-10-03 00:00:00.123456+00'; }],
  ['missing 450', rows => rows.splice(rows.findIndex(row => row.migration_name === MEMORY), 1)],
  ['missing 451', rows => rows.splice(rows.findIndex(row => row.migration_name === PRIOR), 1)],
  ['incorrect 450 checksum', rows => { rows.find(row => row.migration_name === MEMORY).checksum = '0'.repeat(64); }],
  ['incorrect 451 checksum', rows => { rows.find(row => row.migration_name === PRIOR).checksum = '0'.repeat(64); }],

function offlineClient(afterPhase, patch = {}) {
  const calls = [];
  const state = {
    ledger: afterPhase ? after : before, checks: constraints, indexes: afterPhase ? [index] : [],
    priorDefinition, priorReady: true, retiredAbsent: true, newAbsent: true,
    owner: true, locks: 0, readOnly: 'on', ...patch,
  };
  return {
    calls,
    async query(sql) {
      calls.push(sql);
      if (sql === 'BEGIN ISOLATION LEVEL REPEATABLE READ READ ONLY' || sql === 'ROLLBACK' || sql.startsWith('SET LOCAL ')) return { rows: [] };
      if (sql === 'SHOW transaction_read_only') return { rows: [{ transaction_read_only: state.readOnly }] };
      if (sql === owner.LEDGER_SQL) return { rows: clone(state.ledger) };
      if (sql === owner.CHECK_SQL) return { rows: clone(state.checks) };
      if (sql === owner.INDEX_SQL) return { rows: clone(state.indexes) };
      if (sql === owner.LOCK_SQL) return { rows: [{ count: state.locks }] };
      if (sql.startsWith('SELECT pg_get_indexdef(')) return { rows: [{ definition: state.priorDefinition }] };
      if (sql.startsWith('SELECT indisvalid AND indisready')) return { rows: [{ ready: state.priorReady }] };
      if (sql.startsWith("SELECT to_regclass('public.meri_agent_runs')")) return { rows: [{ absent: state.retiredAbsent }] };
      if (sql.startsWith("SELECT to_regclass('public.mail_messages_tenant_connection_active_idx')")) return { rows: [{ absent: state.newAbsent }] };
      if (sql.startsWith('SELECT relname,pg_has_role(')) return { rows: [{ relname: '_prisma_migrations', owner: state.owner }, { relname: 'mail_messages', owner: state.owner }] };
      throw new Error('Unexpected offline query');
    },
  };
}

test('readState performs bounded read-only before/after checks and always rolls back', async () => {
  for (const afterPhase of [false, true]) {
    const client = offlineClient(afterPhase);
    const actual = await owner.readState(client, offlineInput, manifest, afterPhase);
    assert.equal(actual.ledger.length, afterPhase ? 609 : 608);
    assert.equal(client.calls[0], 'BEGIN ISOLATION LEVEL REPEATABLE READ READ ONLY');
    assert.equal(client.calls.at(-1), 'ROLLBACK');
    assert.ok(client.calls.includes("SET LOCAL statement_timeout='5s'"));
    assert.ok(client.calls.every(sql => /^(?:BEGIN|SET LOCAL|SHOW|SELECT|ROLLBACK)\b/.test(sql)));
  }
});

for (const [label, patch] of [
  ['non-read-only transaction', { readOnly: 'off' }],
  ['incorrect prior index definition', { priorDefinition: priorDefinition + ' ' }],
  ['prior index not valid and ready', { priorReady: false }],
  ['retired tables present', { retiredAbsent: false }],
  ['new index already present', { newAbsent: false }],
  ['non-owner connection', { owner: false }],
  ['conflicting lock', { locks: 1 }],
  ['unvalidated memory constraint', { checks: constraints.map((row, i) => i ? row : { ...row, validated: false }) }],
  ['altered memory constraint', { checks: constraints.map((row, i) => i ? row : { ...row, definition: row.definition + ' ' }) }],
]) test(`readState rejects ${label} and rolls back`, async () => {
  const client = offlineClient(false, patch);
  await assert.rejects(() => owner.readState(client, offlineInput, manifest, false));
  assert.equal(client.calls.at(-1), 'ROLLBACK');
});
