export function normalizeDefinition(value){
 const copy=structuredClone(value);
 for(const key of ['volumes','placementConstraints'])if(copy[key]===undefined)copy[key]=[];
 for(const container of copy.containerDefinitions??[]){
  if(container.cpu===undefined)container.cpu=0;
  for(const key of ['portMappings','mountPoints','volumesFrom','systemControls'])if(container[key]===undefined)container[key]=[];
 }
 return copy;
}
function persistReceipt(receipt){

 const fd=openSync(temp,'w');try{writeFileSync(fd,JSON.stringify(receipt,null,2)+'\n');fsyncSync(fd);}finally{closeSync(fd);}
 renameSync(temp,file);
}

export async function run(approved,confirmation){

 assert.equal(planHash,fingerprint(body));assert.equal(approved,planHash);
 const age=Date.now()-Date.parse(plan.preparedAt);assert.ok(age>=0&&age<3600000,'PLAN_EXPIRED');
 assert.equal(plan.runnerOperatorSha256,hash(readFileSync(ROOT+'/forward452-task.mjs')));
 const c=await context();assert.equal(fingerprint(c.live.baseline),fingerprint(plan.apiBaseline));assert.equal(fingerprint(taskInput(c.migrate)),plan.migrationTemplateHash);
 assert.equal(c.scriptHash,plan.runnerSha256);assert.equal(c.inputHash,plan.inputSha256);assert.equal(fingerprint(c.def),plan.taskDefinitionHash);assert.equal(fingerprint(c.live.service.networkConfiguration),plan.networkHash);assert.equal(c.imageProof.image,plan.image);

  const request={cluster:'capiro-dev',taskDefinition:definition,launchType:'FARGATE',platformVersion:c.live.service.platformVersion,networkConfiguration:c.live.service.networkConfiguration,count:1,startedBy:'meri-c1-'+plan.operationId.slice(0,24),clientToken:plan.operationId};

  do{await delay(5000);const r=aws(['ecs','describe-tasks','--cluster','capiro-dev','--tasks',taskArn]);assert.equal(r.failures?.length??0,0);assert.equal(r.tasks?.length,1);task=r.tasks[0];assert.equal(task.taskDefinitionArn,definition);assert.equal(task.containers.length,1);assert.equal(task.containers[0].name,CONTAINER);if(task.containers[0].imageDigest)assert.equal(task.containers[0].imageDigest,plan.image.split('@')[1]);}while(task.lastStatus!=='STOPPED'&&Date.now()<deadline);
  assert.equal(task.lastStatus,'STOPPED','TASK_TIMEOUT_RECONCILE');
  assert.equal(task.containers[0].imageDigest,plan.image.split('@')[1],'STOPPED_TASK_DIGEST_UNVERIFIED');

}
