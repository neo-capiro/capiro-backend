function Assert-OwnedPath([string]$value, [string]$expected) {
  $actual = [IO.Path]::GetFullPath($value).TrimEnd('\','/')
  $fixed = [IO.Path]::GetFullPath($expected).TrimEnd('\','/')
  if (-not $actual.Equals($fixed,[StringComparison]::OrdinalIgnoreCase) -or
      -not $actual.StartsWith($taskRoot + '\',[StringComparison]::OrdinalIgnoreCase)) {
    throw 'Refusing path outside exact task-5 ownership'
  }
  $cursor = $actual
  while ($cursor -and $cursor.Length -ge $taskRoot.Length) {
    if (Test-Path -LiteralPath $cursor) {
      if (((Get-Item -LiteralPath $cursor -Force).Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0) {
        throw 'Refusing a reparse point in the task-owned cluster path'
      }
    }
    if ($cursor.Equals($taskRoot,[StringComparison]::OrdinalIgnoreCase)) { break }
    $cursor = [IO.Path]::GetDirectoryName($cursor)
  }
  return $actual
}
function Git-Bytes([string[]]$gitArgs) {
  $si = New-Object Diagnostics.ProcessStartInfo
  $si.FileName=$gitExe; $si.WorkingDirectory=$source; $si.UseShellExecute=$false
  $si.CreateNoWindow=$true; $si.RedirectStandardOutput=$true; $si.RedirectStandardError=$true
  foreach($a in $gitArgs) { $si.ArgumentList.Add($a) }
  $p = New-Object Diagnostics.Process; $p.StartInfo=$si
  if (-not $p.Start()) { throw 'Could not start read-only git source inspection' }
  $err = $p.StandardError.ReadToEndAsync(); $buffer = New-Object IO.MemoryStream
  $p.StandardOutput.BaseStream.CopyTo($buffer); $p.WaitForExit()
  if ($p.ExitCode -ne 0) { throw ('Git source inspection failed: ' + $err.GetAwaiter().GetResult()) }
  return ,$buffer.ToArray()
}
function Git-Text([string[]]$gitArgs) { return $utf8.GetString((Git-Bytes $gitArgs)) }
function Sha([byte[]]$bytes) {
  $hash = [Security.Cryptography.SHA256]::Create()
  try { return ([BitConverter]::ToString($hash.ComputeHash($bytes))).Replace('-','').ToLowerInvariant() }
  finally { $hash.Dispose() }
}

function Add-SourceUnit([string]$name,[string]$relative,[string]$selection='whole-file') {
  [byte[]]$original = Git-Bytes @('show',($sourceHead + ':' + $relative))
  $text = $utf8.GetString($original); $selected = $text
  if ($selection -eq 'identity-without-unavailable-vector-extension') {
    $needle = 'CREATE EXTENSION IF NOT EXISTS "vector";' + "`n"
    if (($text.Split(@($needle),[StringSplitOptions]::None)).Count -ne 2) { throw 'Identity vector omission is no longer exact' }
    $selected = $text.Replace($needle,'')
  } elseif ($selection -eq 'ethical-wall-ddl-before-unrelated-embedding-backfill') {
    $marker = '-- 3. context_embeddings.client_id parity backfill'
    if (($text.Split(@($marker),[StringSplitOptions]::None)).Count -ne 2) { throw 'Ethical-wall source boundary is no longer exact' }
    $selected = $text.Substring(0,$text.IndexOf($marker,[StringComparison]::Ordinal))
  } elseif ($selection -eq 'exact-client-composite-unique-index-only') {
    $needle = 'CREATE UNIQUE INDEX "clients_tenant_id_key" ON "clients"("tenant_id", "id");'
    if (($text.Split(@($needle),[StringSplitOptions]::None)).Count -ne 2) { throw 'Client composite-index source is no longer exact' }
    $selected = $needle + "`n"
  } elseif ($selection -ne 'whole-file') { throw 'Unknown SQL source selection' }
  [byte[]]$body = $utf8.GetBytes($selected)
  $blob = (Git-Text @('rev-parse',($sourceHead + ':' + $relative))).Trim()
  $units.Add([PSCustomObject]@{ name=$name; path=$relative; gitBlob=$blob; selection=$selection;
    originalSha256=(Sha $original); originalBytes=$original.Length; selectedSha256=(Sha $body);
    selectedBytes=$body.Length; body=$body })
}

function Child([string]$exe,[string[]]$argv,[string]$stdin='',[int]$limitSeconds=60) {
  $si=New-Object Diagnostics.ProcessStartInfo; $si.FileName=$exe; $si.WorkingDirectory=$workspace
  $si.UseShellExecute=$false; $si.CreateNoWindow=$true; $si.RedirectStandardOutput=$true
  $si.RedirectStandardError=$true; $si.RedirectStandardInput=$true
  foreach($a in $argv) { $si.ArgumentList.Add($a) }
  $p=New-Object Diagnostics.Process; $p.StartInfo=$si
  if(-not $p.Start()) { throw 'Could not start approved task-owned child' }
  $out=$p.StandardOutput.ReadToEndAsync(); $err=$p.StandardError.ReadToEndAsync()
  if($stdin) { $p.StandardInput.Write($stdin) }; $p.StandardInput.Close()
  $watch=[Diagnostics.Stopwatch]::StartNew()
  try {
    while(-not $p.WaitForExit(1000)) {
      Assert-Capacity
      if($watch.Elapsed.TotalSeconds -gt $limitSeconds) { throw 'Task-owned child exceeded bounded execution time' }
    }
  } catch { if(-not $p.HasExited) { $p.Kill($true); $p.WaitForExit() }; throw }
  # A detached descendant can inherit a redirected pipe even after its direct
  # parent exits. Never wait indefinitely for EOF after the bounded process wait.
  if(-not $out.Wait(5000) -or -not $err.Wait(5000)) { throw 'Task-owned child output drain exceeded bounded execution time' }
  $text=Redact ($out.GetAwaiter().GetResult()+$err.GetAwaiter().GetResult())
  [IO.File]::AppendAllText((Join-Path $evidenceDir 'commands.log'),("`nEXEC "+[IO.Path]::GetFileName($exe)+" exit="+$p.ExitCode+"`n"+$text),$utf8)
  if($p.ExitCode -ne 0) { throw ('Task-owned '+[IO.Path]::GetFileName($exe)+' failed; sanitized commands.log retained') }
  Assert-Capacity
  return $text
}

function Start-OwnedPostgres {
  # PostgreSQL outlives pg_ctl. Do not give that descendant a captured pipe whose
  # EOF the runner would await before it can reach its own exact-target cleanup.
  # Output contains no credentials; postgres itself logs to the reviewed -l path.
  $si=New-Object Diagnostics.ProcessStartInfo
  $si.FileName=Join-Path $pgBin 'pg_ctl.exe'; $si.WorkingDirectory=$workspace
  $si.UseShellExecute=$false; $si.CreateNoWindow=$true
  foreach($a in @('-w','-t','20','-D',$dataDir,'-l',$serverLog,'start')) { $si.ArgumentList.Add($a) }
  $p=New-Object Diagnostics.Process; $p.StartInfo=$si
  if(-not $p.Start()) { throw 'Could not start exact owned PostgreSQL control process' }
  $watch=[Diagnostics.Stopwatch]::StartNew()
  try {
    while(-not $p.WaitForExit(1000)) {
      Assert-Capacity
      if($watch.Elapsed.TotalSeconds -gt 30) { throw 'Exact owned PostgreSQL start exceeded bounded execution time' }
    }
  } catch { if(-not $p.HasExited) { $p.Kill($true); $p.WaitForExit() }; throw }
  [IO.File]::AppendAllText((Join-Path $evidenceDir 'commands.log'),("`nEXEC pg_ctl.exe start (uncaptured daemon output) exit="+$p.ExitCode+"`n"),$utf8)
  if($p.ExitCode -ne 0) { throw 'Exact owned PostgreSQL start failed' }
  Assert-Capacity
}

  $env:OBJECTIVE_PLANNER_NATIVE_OPT_IN='1'; $env:CI_DISPOSABLE_POSTGRES='1'; $env:OBJECTIVE_PLANNER_NATIVE_DATABASE_URL=$databaseUrl
  Write-Output 'Running concrete native suite as capiro_app with no owner membership or bypass.'
  Child $nodeExe @((Join-Path $workspace 'apps\api-strategy\node_modules\jest\bin\jest.js'),
    '--config',$jestConfig,'--runInBand','--runTestsByPath',
    $continuationSpec,$nativeCoreSpec) '' 180 | Out-Null
  [IO.File]::WriteAllText((Join-Path $evidenceDir 'outcome.json'),'{"nativeSuiteExit":0,"sourcePin":"'+$sourceHead+'","cleanupPending":true}', $utf8)
} finally {
  foreach($name in $envNames) {
    if($null -eq $savedEnv[$name]) { Remove-Item -LiteralPath ('Env:'+ $name) -ErrorAction SilentlyContinue }
    else { [Environment]::SetEnvironmentVariable($name,$savedEnv[$name],'Process') }
  }
  if($created -and (Test-Path -LiteralPath $pwFile)) { Remove-Item -LiteralPath $pwFile -Force }
  if($created -and (Test-Path -LiteralPath $dataDir)) {
    try {
      Assert-OwnedPath $nativeRoot (Join-Path $workspace '.objective-planner-native') | Out-Null
      Assert-OwnedPath $dataDir (Join-Path $nativeRoot 'postgres-data') | Out-Null
      if([IO.File]::ReadAllText((Join-Path $nativeRoot 'run-owner.txt')) -cne $runId) { throw 'Ownership token mismatch; refusing cluster stop' }
      # No service name or generic postgres stop command is used.
      & (Join-Path $pgBin 'pg_ctl.exe') -D $dataDir status *> $null
      if($LASTEXITCODE -eq 0) {
        & (Join-Path $pgBin 'pg_ctl.exe') -w -t 20 -D $dataDir stop -m fast *> $null
        if($LASTEXITCODE -ne 0) { throw 'Exact owned cluster did not stop' }
      }
      & (Join-Path $pgBin 'pg_ctl.exe') -D $dataDir status *> $null
      if($LASTEXITCODE -eq 0) { throw 'Exact owned cluster still runs' }
      if(@([Net.NetworkInformation.IPGlobalProperties]::GetIPGlobalProperties().GetActiveTcpListeners()|Where-Object Port -eq $port).Count) { throw 'Owned port still listens' }
      if(Test-Path -LiteralPath (Join-Path $dataDir 'postmaster.pid')) {
        $recorded=[int]([IO.File]::ReadAllLines((Join-Path $dataDir 'postmaster.pid'))[0])
        if(Get-Process -Id $recorded -ErrorAction SilentlyContinue) { throw 'Recorded postmaster PID still exists' }
      }
    } catch { $cleanupProblem='Owned cluster cleanup is uncertain; exact data retained: '+$_.Exception.Message }
  }
  if($created -and (Test-Path -LiteralPath $serverLog)) {
    try { [IO.File]::WriteAllText((Join-Path $evidenceDir 'postgres-sanitized.log'),(Redact ([IO.File]::ReadAllText($serverLog))),$utf8) }
    catch { $cleanupProblem='Could not retain sanitized server log; exact cluster root retained' }
  }
  if($created -and -not $cleanupProblem) {
    Assert-OwnedPath $nativeRoot (Join-Path $workspace '.objective-planner-native') | Out-Null
    Assert-OwnedPath $dataDir (Join-Path $nativeRoot 'postgres-data') | Out-Null
    if([IO.File]::ReadAllText((Join-Path $nativeRoot 'run-owner.txt')) -cne $runId) { throw 'Ownership token mismatch; refusing recursive cleanup' }
    if(Get-ChildItem -LiteralPath $nativeRoot -Recurse -Force | Where-Object { ($_.Attributes -band [IO.FileAttributes]::ReparsePoint) -ne 0 }) {
      throw 'Unexpected reparse point under owned cluster; refusing recursive cleanup'
    }
    Remove-Item -LiteralPath $nativeRoot -Recurse -Force
    [IO.File]::WriteAllText((Join-Path $evidenceDir 'cleanup.json'),'{"exactClusterRemoved":true,"listenerRemoved":true,"credentialsRetiredWithCluster":true}', $utf8)
  }
  $adminSecret=$null; $appSecret=$null; $databaseUrl=$null; $secrets.Clear()
  if($cleanupProblem) { throw $cleanupProblem }
}
